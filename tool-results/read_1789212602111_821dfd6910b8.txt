    82→function expiryCountdown(expiresAt: string | null, now: number | null): string | null {
    83→  if (!expiresAt || now === null) return null;
    84→  const ms = new Date(expiresAt).getTime() - now;
    85→  if (ms <= 0) return null;
    86→  const mins = Math.floor(ms / 60000);
    87→  if (mins < 1) return "<1m left";
    88→  if (mins < 60) return `${mins}m left`;
    89→  return `${Math.floor(mins / 60)}h ${mins % 60}m left`;
    90→}
