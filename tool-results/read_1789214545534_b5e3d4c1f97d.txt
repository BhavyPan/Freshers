    49→/** Entry-heatmap intensity ramp — quiet → busy (purple → fuchsia → amber). */
    50→function heatClass(count: number): string {
    51→  if (count <= 0) return "bg-purple-500/[0.06] border-purple-500/10";
    52→  if (count === 1) return "bg-purple-500/30 border-purple-400/25";
    53→  if (count <= 3) return "bg-purple-500/55 border-purple-300/30";
    54→  if (count <= 6) return "bg-fuchsia-500/75 border-fuchsia-300/40 shadow-[0_0_8px_rgba(217,70,239,0.25)]";
    55→  return "bg-amber-400/85 border-amber-300/60 shadow-[0_0_12px_rgba(251,191,36,0.4)]";
    56→}

    58→function EntryHeatmap({ heatmap }: { heatmap: { bucket: string; count: number }[] }) {
    59→  const ROW = 12; // 12 × 15-min = 3 hours per row, 4 rows = 12 hours
    60→  const rows: { bucket: string; count: number }[][] = [];
    61→  for (let i = 0; i < heatmap.length; i += ROW) rows.push(heatmap.slice(i, i + ROW));
    62→  const max = Math.max(1, ...heatmap.map((h) => h.count));
    63→  return (
    64→    <div>
    65→      <div className="space-y-1.5">
    66→        {rows.map((row, ri) => (
    67→          <div key={ri} className="flex items-center gap-1.5">
    68→            <span className="w-14 shrink-0 text-right font-mono text-[9px] uppercase tracking-wide text-purple-200/40">
    69→              {format(new Date(row[0].bucket), "h a")}
    70→            </span>
    71→            <div className="grid flex-1 grid-cols-12 gap-1.5">
    72→              {row.map((cell, ci) => {
    73→                const isNow = ri === rows.length - 1 && ci === row.length - 1;
    74→                return (
    75→                  <div
    76→                    key={cell.bucket}
    77→                    title={`${format(new Date(cell.bucket), "h:mm a")} — ${cell.count} ${cell.count === 1 ? "entry" : "entries"}`}
    78→                    className={cn(
    79→                      "h-6 rounded-[5px] border transition-transform hover:scale-110 hover:z-10",
    80→                      heatClass(cell.count),
    81→                      isNow && "ring-2 ring-purple-300/80 ring-offset-1 ring-offset-[#0e0819]"
    82→                    )}
    83→                  />
    84→                );
    85→              })}
    86→            </div>
    87→          </div>
    88→        ))}
    89→      </div>
    90→      <div className="mt-3 flex flex-wrap items-center justify-between gap-2">
    91→        <p className="text-[10px] text-purple-200/40">
    92→          Each square = 15 minutes · latest square ringed · busiest {max} in a quarter hour
    93→        </p>
    94→        <div className="flex items-center gap-1.5" aria-hidden>
    95→          <span className="text-[9px] uppercase tracking-[0.14em] text-purple-200/45">quiet</span>
    96→          {[0, 1, 3, 6, 9].map((c) => (
    97→            <span key={c} className={cn("h-3 w-3 rounded-[3px] border", heatClass(c))} />
    98→          ))}
    99→          <span className="text-[9px] uppercase tracking-[0.14em] text-purple-200/45">busy</span>
   100→        </div>
   101→      </div>
   102→    </div>
   103→  );
   104→}

   123→function DeptGoals({ depts }: { depts: { dept: string; total: number; checkedIn: number }[] }) {
   124→  const rows = depts.filter((d) => d.total > 0);
   125→  const completeCount = rows.filter((d) => d.checkedIn >= d.total).length;
   126→  return (
   127→    <div>
   128→      <div className="obs-scrollbar max-h-96 space-y-3.5 overflow-y-auto pr-1">
   129→        {rows.map((d) => {
   130→          const pct = Math.min(100, Math.round((d.checkedIn / d.total) * 100));
   131→          const done = d.checkedIn >= d.total;
   132→          return (
   133→            <div key={d.dept} className="group">
   134→              <div className="mb-1.5 flex items-baseline justify-between gap-2">
   135→                <p className="flex min-w-0 items-center gap-1.5 text-[12px] font-bold uppercase tracking-[0.12em] text-purple-100">
   136→                  <span className="truncate">{d.dept}</span>
   137→                  {done && <Trophy className="h-3.5 w-3.5 shrink-0 text-amber-300 drop-shadow-[0_0_6px_rgba(252,211,77,0.8)]" aria-label="Goal complete" />}
   138→                </p>
   139→                <p className="shrink-0 font-mono text-[10px] tabular-nums text-purple-200/55">
   140→                  {d.checkedIn}/{d.total}
   141→                  <span className={cn("ml-2 font-bold", done ? "text-emerald-300" : pct >= 50 ? "text-fuchsia-300" : "text-purple-200/70")}>
   142→                    {pct}%
   143→                  </span>
   144→                </p>
   145→              </div>
   146→              <div
   147→                className={cn(
   148→                  "relative h-2.5 overflow-hidden rounded-full border",
   149→                  done
   150→                    ? "border-emerald-400/50 bg-emerald-500/10 shadow-[0_0_14px_rgba(52,211,153,0.25)]"
   151→                    : "border-purple-500/25 bg-[#0b0517]"
   152→                )}
   153→                role="progressbar"
   154→                aria-valuenow={pct}
   155→                aria-valuemin={0}
   156→                aria-valuemax={100}
   157→                aria-label={`${d.dept}: ${d.checkedIn} of ${d.total} checked in`}
   158→              >
   159→                <motion.div
   160→                  initial={{ width: 0 }}
   161→                  animate={{ width: `${pct}%` }}
   162→                  transition={{ duration: 0.8, ease: [0.22, 1, 0.36, 1] }}
   163→                  className={cn(
   164→                    "relative h-full rounded-full",
   165→                    done
   166→                      ? "bg-gradient-to-r from-emerald-600 via-emerald-400 to-teal-300"
   167→                      : "obs-goal-shimmer bg-gradient-to-r from-violet-800 via-purple-500 to-fuchsia-400"
   168→                  )}
   169→                >
   170→                  {!done && pct > 0 && (
   171→                    <span className="absolute right-0 top-1/2 h-3 w-1 -translate-y-1/2 rounded-full bg-purple-200 shadow-[0_0_8px_rgba(216,180,254,0.9)]" />
   172→                  )}
   173→                </motion.div>
   174→              </div>
   175→            </div>
   176→          );
   177→        })}
   178→      </div>
   179→      <p className="mt-3 text-[10px] text-purple-200/40">
   180→        Goal: every registered junior inside · {completeCount}/{rows.length} departments complete
   181→      </p>
   182→    </div>
   183→  );
   184→}

   186→function KpiCard({
   187→  icon: Icon,
   188→  label,
   189→  value,
   190→  accent,
   191→  delay,
   192→  trend,
   193→}: {
   194→  icon: React.ComponentType<{ className?: string }>;
   195→  label: string;
   196→  value: number | string;
   197→  accent: "purple" | "green" | "amber" | "red";
   198→  delay: number;
   199→  trend?: { text: string; tone: "up" | "flat" } | null;
   200→}) {
   201→  const accents = {
   202→    purple: "border-purple-400/40 text-purple-200 from-purple-600/25 bg-purple-500/10",

   206→  }[accent];

   208→  return (
   209→    <motion.div
   210→      initial={{ opacity: 0, y: 18 }}
   211→      animate={{ opacity: 1, y: 0 }}
   212→      transition={{ delay, duration: 0.45 }}
   213→      className={cn("obs-card-hover relative overflow-hidden rounded-2xl border bg-gradient-to-br to-transparent p-5 transition-all duration-300 hover:-translate-y-0.5 hover:shadow-[0_8px_30px_rgba(147,51,234,0.18)]", accents.split(" ")[0], accents.split(" ")[2], accents.split(" ")[3])}
   214→    >
   215→      <div className="pointer-events-none absolute -right-8 -top-8 h-24 w-24 rounded-full bg-current opacity-[0.06] blur-2xl" />
   216→      <div className="flex items-center justify-between">
   217→        <span className={cn("flex h-9 w-9 items-center justify-center rounded-lg border bg-current/10", accents.split(" ")[0])}>
   218→          <Icon className="h-4.5 w-4.5" />
   219→        </span>
   220→        {trend && (
   221→          <span
   222→            className={cn(
   223→              "flex items-center gap-1 rounded-full border px-2 py-0.5 text-[9px] font-bold uppercase tracking-[0.12em]",
   224→              trend.tone === "up"
   225→                ? "border-emerald-400/40 bg-emerald-500/10 text-emerald-300"
   226→                : "border-purple-500/25 bg-purple-500/8 text-purple-200/60"

   229→          >
   230→            <TrendingUp className={cn("h-3 w-3", trend.tone === "flat" && "opacity-50")} />
   231→            {trend.text}
   232→          </span>
   233→        )}
   234→      </div>
   235→      <p className="font-display mt-4 text-3xl font-black text-purple-50">
   236→        <AnimatedNumber value={value} />
   237→      </p>
   238→      <p className="mt-1 text-[11px] font-semibold uppercase tracking-[0.18em] text-purple-200/60">{label}</p>
   239→    </motion.div>
   240→  );
   241→}

   243→export function DashboardView() {
   244→  const { toast } = useToast();
   245→  const { data, isLoading } = useQuery({
   246→    queryKey: ["stats"],
   247→    queryFn: () => api.stats(),
   248→    refetchInterval: 4000,
   249→    refetchIntervalInBackground: false,
   250→  });
   251→
   252→  const stats = data?.stats;
   253→  const recent = data?.recent ?? [];
   254→
   255→  // ---- department goal celebrations (confetti on a dept hitting 100%) ----
   256→  const [goalBurst, setGoalBurst] = useState<string | null>(null);
   257→  const doneDeptsRef = useRef<Set<string> | null>(null);
   258→  const deptKey = JSON.stringify(stats?.studentsByDept ?? []);
   259→
   260→  useEffect(() => {
   261→    const depts = JSON.parse(deptKey) as { dept: string; total: number; checkedIn: number }[];
   262→    if (depts.length === 0) return;
   263→    // first run: hydrate the already-complete set from localStorage (no celebration)
   264→    if (doneDeptsRef.current === null) {
   265→      let stored: unknown = [];
   266→      try {
   267→        stored = JSON.parse(window.localStorage.getItem("obs-goals-done") ?? "[]");
   268→      } catch {
   269→        stored = [];
   270→      }
   271→      doneDeptsRef.current = new Set(
   272→        Array.isArray(stored) ? stored.filter((s): s is string => typeof s === "string") : []
   273→      );
   274→    }
   275→    const complete = depts.filter((d) => d.total > 0 && d.checkedIn >= d.total).map((d) => d.dept);
   276→    const fresh = complete.find((d) => !doneDeptsRef.current!.has(d));
   277→    if (!fresh) return;
   278→    for (const d of complete) doneDeptsRef.current!.add(d);
   279→    try {
   280→      window.localStorage.setItem("obs-goals-done", JSON.stringify([...doneDeptsRef.current]));
   281→    } catch {
   282→      /* private mode — celebration still fires, just not persisted */
   283→    }
   284→    setGoalBurst(fresh);
   285→    playFeedback("granted");
   286→    toast({
   287→      title: `${fresh} squad is all in! 🎉`,
   288→      description: `Every registered ${fresh} junior has checked in — goal complete.`,
   289→    });
   290→    const t = setTimeout(() => setGoalBurst(null), 2600);
   291→    return () => clearTimeout(t);
   292→  }, [deptKey, toast]);
   293→
   294→  const timelineData = (stats?.timeline ?? []).map((t) => ({
   295→    time: format(new Date(t.bucket), "HH:mm"),
   296→    count: t.count,
   297→  }));
   298→
   299→  const deptData = (stats?.studentsByDept ?? []).slice(0, 8).map((d) => ({
   300→    dept: d.dept,
   301→    total: d.total,
   302→    checkedIn: d.checkedIn,
   303→  }));
   304→
   305→  if (isLoading || !stats) {
   306→    return (
   307→      <div className="space-y-6">
   308→        <div className="grid grid-cols-2 gap-4 xl:grid-cols-4">
   309→          {Array.from({ length: 4 }).map((_, i) => (
   310→            <Skeleton key={i} className="h-36 rounded-2xl bg-purple-500/10" />
   311→          ))}
   312→        </div>
   313→        <Skeleton className="h-72 rounded-2xl bg-purple-500/10" />
   314→        <Skeleton className="h-64 rounded-2xl bg-purple-500/10" />
   315→      </div>
   316→    );
   317→  }
   318→
   319→  return (
   320→    <div className="space-y-6">
   321→      {/* goal celebration overlay */}
   322→      <AnimatePresence>
   323→        {goalBurst && (
   324→          <motion.div
   325→            key={goalBurst}
   326→            initial={{ opacity: 0 }}
   327→            animate={{ opacity: 1 }}
   328→            exit={{ opacity: 0 }}
   329→            className="pointer-events-none fixed inset-0 z-50 flex items-center justify-center"
   330→            aria-live="polite"
   331→          >
   332→            <SparkleBurst count={34} />
   333→            <motion.div
   334→              initial={{ scale: 0.6, y: 24, opacity: 0 }}
   335→              animate={{ scale: 1, y: 0, opacity: 1 }}
   336→              exit={{ scale: 0.85, y: -18, opacity: 0 }}
   337→              transition={{ type: "spring", stiffness: 260, damping: 18 }}
   338→              className="flex items-center gap-3 rounded-2xl border border-amber-300/50 bg-[#140b26]/95 px-6 py-4 shadow-[0_0_60px_rgba(251,191,36,0.35)]"
   339→            >
   340→              <PartyPopper className="h-7 w-7 text-amber-300" />
   341→              <div>
   342→                <p className="font-display text-lg font-black text-amber-100">{goalBurst} squad is ALL IN!</p>
   343→                <p className="text-[11px] uppercase tracking-[0.18em] text-purple-200/60">100% checked in · goal complete</p>
   344→              </div>
   345→              <Trophy className="h-6 w-6 text-amber-300 drop-shadow-[0_0_10px_rgba(252,211,77,0.9)]" />
   346→            </motion.div>

   359→          delay={0.08}
   360→          trend={{ text: `+${stats.checkedInLastHour} / 1h`, tone: stats.checkedInLastHour > 0 ? "up" : "flat" }}
   361→        />
   362→        <KpiCard icon={Hourglass} label="Not Arrived" value={stats.notArrived} accent="amber" delay={0.16} />
