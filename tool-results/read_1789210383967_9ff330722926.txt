    36→    36→function timeFmt(iso: string | null) {
    37→    37→  if (!iso) return "—";
    38→    38→  try {
    39→    39→    return format(new Date(iso), "dd MMM, h:mm a");
    40→    40→  } catch {
    41→    41→    return iso;
    42→    42→  }
    43→    43→}
